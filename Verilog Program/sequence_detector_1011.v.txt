module sequence_detector_1011 ( 
    input  wire clk, 
    input  wire reset, 
    input  wire X, 
    output reg  Y 
); 
 
    reg Q1, Q0; 
    reg D1, D0; 
 
    // Next-state logic 
    always @(*) begin 
        D1 = (Q0 & ~X) | (Q1 & ~Q0 & X); 
        D0 = X; 
    end 
 
    // State register 
    always @(posedge clk) begin 
        if (reset) begin 
            Q1 <= 1'b0; 
            Q0 <= 1'b0; 
        end 
        else begin 
            Q1 <= D1; 
            Q0 <= D0; 
        end
         end 
 
    // Output logic (Mealy) 
    always @(*) begin 
        Y = Q1 & Q0 & X; 
    end 
 
endmodule
