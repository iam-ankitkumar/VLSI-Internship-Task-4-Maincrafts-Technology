module mealy_fsm ( 
    input clk, 
    input reset, 
    input X, 
    output reg Y 
); 
 
    reg state; 
 
    // State encoding 
    parameter S0 = 1'b0; 
    parameter S1 = 1'b1; 
 
    // State transition logic 
    always @(posedge clk) begin 
        if (reset) 
            state <= S0; 
        else begin 
            case (state) 
 
                S0: begin 
                    if (X == 1'b0) 
                        state <= S0; 
                    else 
                        state <= S1; 
                end 
 
                S1: begin 
                    if (X == 1'b0) 
                        state <= S1; 
11 
 
                    else 
                        state <= S0; 
                end 
 
                default: state <= S0; 
 
            endcase 
        end 
    end 
 
    // Mealy output logic 
    always @(*) begin 
        case (state) 
 
            S0: Y = 1'b0; 
 
            S1: begin 
                if (X == 1'b1) 
                    Y = 1'b1; 
                else 
                    Y = 1'b0; 
            end 
 
            default: Y = 1'b0; 
 
        endcase 
    end 
 
endmodule
